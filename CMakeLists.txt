cmake_minimum_required(VERSION 3.20)
project(RockSim VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(ROCKSIM_BUILD_TESTS "Build the GoogleTest suite" ON)
option(ROCKSIM_BUILD_CLI "Build the rocksim-cli tool" ON)
option(ROCKSIM_BUILD_BENCHMARKS "Build Google Benchmark targets" OFF)
option(ROCKSIM_SANITIZE "Build with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)

add_library(rocksim_warnings INTERFACE)
if(MSVC)
  target_compile_options(rocksim_warnings INTERFACE /W4 /permissive-)
else()
  target_compile_options(rocksim_warnings INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wnon-virtual-dtor)
endif()
if(ROCKSIM_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined)
endif()

add_subdirectory(engine)

if(EMSCRIPTEN)
  add_subdirectory(bindings/wasm)
else()
  if(ROCKSIM_BUILD_CLI)
    add_subdirectory(cli)
  endif()
  if(ROCKSIM_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
  endif()
  if(ROCKSIM_BUILD_BENCHMARKS)
    add_subdirectory(bench)
  endif()
endif()
